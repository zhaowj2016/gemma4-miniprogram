Page({
  data: {
    allProducts: [
      { id: 1, name: '智能声控降噪运动耳机', price: '299.00', sales: 120, image: 'https://images.unsplash.com/photo-1523275335684-37898b6baf30?w=300' },
      { id: 2, name: '极简白时尚保温杯', price: '89.00', sales: 450, image: 'https://images.unsplash.com/photo-1574269909862-7e1d70bb8078?w=300' },
      { id: 3, name: '多功能无线快充移动电源', price: '159.00', sales: 80, image: 'https://images.unsplash.com/photo-1622445262465-2481c4574875?w=300' }
    ],
    products: []
  },
  onLoad: function() {
    this.setData({
      products: this.data.allProducts
    });
  },
  onSearchInput: function(e) {
    const val = e.detail.value.toLowerCase();
    const filtered = this.data.allProducts.filter(item => item.name.toLowerCase().includes(val));
    this.setData({
      products: filtered
    });
  },
  onProductTap: function(e) {
    const id = e.currentTarget.dataset.id;
    wx.showToast({
      title: '点击商品ID: ' + id,
      icon: 'none'
    });
  }
})
