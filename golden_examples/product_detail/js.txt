Page({
  data: {},
  addToCart: function() {
    wx.showToast({
      title: '已加入购物车',
      icon: 'success'
    });
  },
  buyNow: function() {
    wx.showModal({
      title: '提示',
      content: '确定要立即购买吗？',
      success: function(res) {
        if (res.confirm) {
          wx.showToast({
            title: '下单成功',
            icon: 'success'
          });
        }
      }
    });
  }
})
