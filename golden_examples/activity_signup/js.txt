Page({
  data: {},
  onSubmit: function(e) {
    const formData = e.detail.value;
    if (!formData.name || !formData.phone) {
      wx.showToast({
        title: '请填写必填项',
        icon: 'error'
      });
      return;
    }
    wx.showToast({
      title: '报名成功',
      icon: 'success'
    });
  }
})
